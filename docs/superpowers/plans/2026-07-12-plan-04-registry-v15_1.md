**Interfaces (Produces):**

```python
@dataclass(frozen=True)
class StrategyManifest:
    id: str                    # unique slug, e.g. "silver_bullet"
    version: str               # e.g. "14.4.2"
    class_path: str            # "src.strategies.models.silver_bullet:SilverBullet"
    family: str                # "smc" | "statistical" | "physics" | ...
    timeframe: str             # "M5" | "M15" | "H1"
    requires: tuple[str, ...]  # FeatureBus resource names
    status: str                # "research" | "demo" | "live"
    priority: int = 50

class ManifestError(ValueError): ...

def load_manifest(path) -> StrategyManifest          # single YAML file
def load_manifests(dir_path) -> list[StrategyManifest]  # *.yaml sorted by filename; duplicate ids -> ManifestError
```

Validation in `load_manifest` (each failure raises `ManifestError` naming the file and field): required keys present; `status` in the 3-value set; `timeframe` in {"M5","M15","H1"}; `class_path` contains exactly one ":"; `requires` a list of strings (may be empty); `priority` an int. No class import here (that's the registry's job).

```python
# events.py additions (registry pattern identical to existing events):
@_register @dataclass(frozen=True)
class StrategyActivated(Event):
    name = "StrategyActivated"; strategy_id: str = ""; version: str = ""; family: str = ""; timeframe: str = ""
@_register @dataclass(frozen=True)
class StrategySuspended(Event):
    name = "StrategySuspended"; strategy_id: str = ""; version: str = ""; reason: str = ""

class RegistryError(ValueError): ...

class StrategyRegistry:
    def __init__(self, manifests, params_by_id, logger, publish=None, feature_bus=None):
        # params_by_id: dict id -> config block (config.yaml strategies.<id>)
        # publish: callable(Event) or None; feature_bus: for requires-validation or None
    def load_all(self) -> None
        # for each manifest: import class (importlib, split ":"), instantiate
        # cls(params_by_id.get(id, {}), logger). Import/instantiate failure -> RegistryError.
        # If feature_bus given: every name in requires must exist in its registry
        # (bus._registry keys) else RegistryError naming the missing resource.
        # State: REGISTERED -> LOADED.
    def activate_eligible(self) -> None
        # LOADED -> ACTIVE for manifests with status in ("demo","live") AND
        # params.get('enabled', True) is not False. Publishes StrategyActivated.
        # research-status manifests stay LOADED (present, not trading).
    def enable(self, strategy_id) -> str   # LOADED|SUSPENDED -> ACTIVE (publish); returns human message
    def disable(self, strategy_id, reason="operator") -> str  # ACTIVE -> SUSPENDED (publish StrategySuspended)
        # unknown id or illegal transition -> returns error message string (no raise: Telegram-facing)
    def active_instances(self) -> list      # ACTIVE instances, ordered by (priority, id)
    def instance_of(self, strategy_id)      # -> instance or None
    def state_of(self, strategy_id) -> str  # "REGISTERED"|"LOADED"|"ACTIVE"|"SUSPENDED"
    def report(self) -> list[dict]          # per strategy: id, version, family, tf, status, state, priority
```

```python
    def _init_strategies(self):
```
